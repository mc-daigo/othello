  ["none", "none", "none", "none", "none", "none", "none", "none"],
  ["none", "none", "none", "none", "none", "none", "none", "none"],
  ["none", "none", "none", "none", "none", "none", "none", "none"],

  ["none", "none", "none", "black", "white", "none", "none", "none"],
  ["none", "none", "none", "none", "none", "none", "none", "none"],
  ["none", "none", "none", "none", "none", "none", "none", "none"],
  ["none", "none", "none", "none", "none", "none", "none", "none"],
];

const [squares, setSquares] = useState<string[][]>(SQUARES);
const [isAnimating, setIsAnimating] = useState(false);
const [isBlack, setIsBlack] = useState(true);

const changeSquares = useCallback(
  (row: number, column: number) => {
    setSquares((prevSquares) => {
      const newSquares = prevSquares.map((r, rowIndex) =>
        r.map((stone, colIndex) =>
          rowIndex === row && colIndex === column
            ? `${isBlack ? "black" : "white"} placing`
            : stone
        )
      );
      return newSquares;
    });

    // アニメーションが終わった後に状態を更新
    setTimeout(() => {
      setSquares((prevSquares) => {
        const updatedSquares = prevSquares.map((r, rowIndex) =>
          r.map((stone, colIndex) => {
            if (rowIndex === row && colIndex === column) {
              return isBlack ? "black" : "white"; // `placing`を除去
            }
            return stone;
          })
        );
        setIsAnimating(false);
        setIsBlack(!isBlack); // 次のターンの色に切り替え
        return updatedSquares;
      });
    }, 300); // アニメーション時間と一致させる
  },
  [isBlack]
);

const clickSquare = useCallback(
  (e: React.MouseEvent<HTMLLIElement>) => {
    const liElement = e.currentTarget;
    const row = Number(liElement.dataset.row);
    const column = Number(liElement.dataset.column);

    // 既に石が置かれている、またはアニメーション中なら何もしない
    if (isAnimating || squares[row][column] !== "none") {
      return;
    }

    setIsAnimating(true); // アニメーション中フラグを立てる
    changeSquares(row, column);
  },
  [squares, isAnimating, changeSquares]
);

              data-column={columnIndex}

                  .map((cls) => styles[cls])

              ></p>
            </li>
          ))}
        </ul>

CSS モジュールでは、クラス名を補完するには以下の方法が必要です：

解決策
1. stylesを活用してクラスを動的に構築する
CSS モジュールでクラス名を補完させるには、stylesオブジェクトを使って各クラス名を個別に解決し、最終的にスペースで結合する必要があります。

tsx

    .join(" ")} // 配列をスペースで結合してクラス名文字列を生成
  data-stone={stone}
>
</p>
これにより、styles["black"]やstyles["placing"]が解決され、適切なNext.jsの補完済みクラス名がclassNameに反映されます。

CSS モジュールの仕組み
CSS モジュールファイルを作成
例えば、styles.module.css というファイルに以下のようなスタイルを定義するとします。

css

.black {

}

.placing {
  animation: fadeIn 0.5s ease-in-out;
}
Next.js による処理
この .module.css ファイルを React コンポーネントでインポートすると、Next.js はクラス名をユニークな形式に変換して styles オブジェクトを作成します。

tsx
コードをコピーする

このインポートによって、styles オブジェクトが自動的に生成されます。

生成されるオブジェクトの例:

javascript
コードをコピーする
const styles = {
  black: "black_123abc",   // ユニークなクラス名
  placing: "placing_456def", // ユニークなクラス名
};
元のクラス名（例: .black, .placing）をキーとして持ち、
Next.js が生成したユニークなクラス名（例: black_123abc, placing_456def）を値として持つオブジェクトです。
className に利用
styles.black や styles.placing を React の className 属性に渡すことで、Next.js が自動生成したユニークなクラス名を要素に適用します。

tsx

これにより、HTML は次のようにレンダリングされます。

html
コードをコピーする
<p class="black_123abc">Black Stone</p>
ユニークなクラス名が必要な理由
CSS モジュールでは、各クラス名をユニークな形式に変換することで、グローバルな名前空間の衝突を防止します。これにより、他のコンポーネントやスタイルシートで同じクラス名が定義されていても問題が起きません。

ポイント
styles は Next.js が自動生成するオブジェクトで、手動で作成する必要はありません。
元のクラス名をキーとして持ち、ユニークなクラス名を値としてマッピングします。
className 属性に利用することで、ユニークなクラス名を簡単に適用できます。
これにより、モジュール化されたスタイルが安全かつ簡単に使用できるようになっています！
